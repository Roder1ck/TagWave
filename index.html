How TagWave works

This app runs entirely in your browser. It sends audio fingerprints to Shazam's public API (via a CORS proxy) to identify tracks, then lets you edit and export metadata. For actually writing tags to your audio files, you need the companion Python script below — browsers can't write ID3/FLAC tags directly.

Deploy to GitHub Pages

1

Create a new repository

Go to github.com → New repository → name it tagwave (or anything). Initialize with a README.

2

Upload index.html

Upload this index.html file to your repo root. You can also add tagger.py there.

3

Enable GitHub Pages

Settings → Pages → Source: Deploy from branch → Branch: main → folder: / (root) → Save. Your site will be live at https://yourusername.github.io/tagwave

Python tag writer — tagger.py

Download your metadata JSON from TagWave, then run this script to write tags and sort files into genre folders.

# tagger.py — pip install shazamio mutagen import json, shutil, asyncio from pathlib import Path from mutagen.id3 import ID3, TIT2, TPE1, TALB, TCON, TBPM, TDRC, TPE4 from mutagen.flac import FLAC from shazamio import Shazam # ── Config ───────────────────────────────────── INPUT_DIR = Path("./music") # your audio files OUTPUT_DIR = Path("./sorted") # sorted output META_JSON = Path("./metadata.json") # exported from TagWave # ── Genre → folder map ────────────────────────── GENRE_MAP = { "Deep House": "House", "Tech House": "House", "Progressive House": "House", "Afro House": "House", "Melodic Techno": "Techno", "Hard Techno": "Techno", "Detroit Techno": "Techno", "Industrial": "Techno", "Drum and Bass": "DnB", "Liquid DnB": "DnB", "Jungle": "DnB", "Dubstep": "Bass", "Riddim": "Bass", "Trap": "Bass", "UK Bass": "Bass", "Trance": "Trance", "Psytrance": "Trance", "Hip-Hop": "Hip-Hop", "Rap": "Hip-Hop", "Electronic": "Electronic", "EDM": "Electronic", } async def recognize_file(path): shazam = Shazam() out = await shazam.recognize(str(path)) track = out.get("track", {}) return { "title": track.get("title", ""), "artist": track.get("subtitle", ""), "album": track.get("sections", [{}])[0].get("metadata", []), "genre": track.get("genres", {}).get("primary", ""), } def write_mp3_tags(path, meta): tags = ID3(path) tags["TIT2"] = TIT2(text=[meta["title"]]) tags["TPE1"] = TPE1(text=[meta["artist"]]) tags["TCON"] = TCON(text=[meta["genre"]]) if meta.get("bpm"): tags["TBPM"] = TBPM(text=[meta["bpm"]]) if meta.get("remixedBy"): tags["TPE4"] = TPE4(text=[meta["remixedBy"]]) tags.save() def sort_file(path, genre, folder_override=None): folder = folder_override or GENRE_MAP.get(genre, "Unsorted") dest = OUTPUT_DIR / folder / path.name dest.parent.mkdir(parents=True, exist_ok=True) shutil.copy2(path, dest) return dest # ── Main ──────────────────────────────────────── async def main(): metas = json.loads(META_JSON.read_text()) if META_JSON.exists() else {} for f in INPUT_DIR.rglob("*"): if f.suffix.lower() not in {".mp3",".flac",".wav",".m4a"}: continue meta = metas.get(f.name) or await recognize_file(f) if f.suffix == ".mp3": write_mp3_tags(f, meta) dest = sort_file(f, meta.get("genre",""), meta.get("folderOverride")) print(f"✓ {f.name} → {dest.parent.name}/") asyncio.run(main())

CORS proxy note

Shazam's API doesn't allow direct browser requests. This app uses corsproxy.io as a relay. For production use or high volume, run the Python script directly — it calls Shazam natively without a proxy.
